3. Add `handleDownloadPdf` callback after `handleSourceChange` (around line 95):
```js
const handleDownloadPdf = useCallback(() => {
  setPdfLoading(true);
  const token = typeof window !== 'undefined' ? localStorage.getItem('accessToken') : null;
  const params = new URLSearchParams();
  if (filters.dateFrom)    params.set('dateFrom', filters.dateFrom);
  if (filters.dateTo)      params.set('dateTo', filters.dateTo);
  if (filters.gameId)      params.set('gameId', filters.gameId);
  if (filters.source)      params.set('source', filters.source);
  if (filters.apiSystemId) params.set('apiSystemId', filters.apiSystemId);

  fetch(`${API_URL}/api/monitor/reporte/pdf?${params.toString()}`, {
    headers: { Authorization: `Bearer ${token}` },
  })
    .then(res => {
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      return res.blob();
    })
    .then(blob => {
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `reporte-${filters.dateFrom || 'hoy'}-${filters.dateTo || 'hoy'}.pdf`;
      a.click();
      URL.revokeObjectURL(url);
    })
    .catch(() => toast.error('Error generando PDF'))
    .finally(() => setPdfLoading(false));
}, [filters]);
```
Note: Fetch+blob is used (not window.open) because the endpoint requires an Authorization header — a plain URL opened in a new tab would not carry the token and would receive 401.
