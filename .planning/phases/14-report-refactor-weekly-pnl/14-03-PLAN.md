ISO-week helpers (added in this plan to backend/src/lib/dateUtils.js):
```javascript
// New exports:
import { getISOWeek, getISOWeekYear, startOfISOWeek, addDays } from 'date-fns';
import { fromZonedTime } from 'date-fns-tz';
const VENEZUELA_TIMEZONE = 'America/Caracas';

export { getISOWeek, getISOWeekYear }; // re-export — defensive per assumption A1

/**
 * Returns the Date corresponding to Monday 00:00:00 America/Caracas for the
 * given ISO year + week, in UTC. Per Phase 12 D-04 ISO week boundary.
 */
export function getMondayOfISOWeek(isoYear, isoWeek) {
  // Use date-fns: pick Jan 4 (always in ISO week 1), find its ISO year, then
  // adjust to the requested week, then take startOfISOWeek (Monday).
  const jan4 = new Date(isoYear, 0, 4);
  // Shift by (isoWeek - 1) weeks
  const targetInWeek = addDays(jan4, (isoWeek - 1) * 7);
  const mondayLocal = startOfISOWeek(targetInWeek); // Monday 00:00 local
  // Convert Caracas-local Monday-00:00 to UTC instant
  return fromZonedTime(mondayLocal, VENEZUELA_TIMEZONE);
}
```
