import apiClient from './client'; // existing axios instance

const pnlAPI = {
  getWeeklyPnl: async (params) => {
    const response = await apiClient.get('/reportes/pnl/semanal', { params });
    return response.data;
  },
  getExcelUrl: (params) => {
    const q = new URLSearchParams(params).toString();

};
